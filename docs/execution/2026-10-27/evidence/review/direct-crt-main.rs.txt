extern crate exact_transcendentals;
use exact_transcendentals::cram_ct::{CramCiphertext,CRAM_DIVISION_COUNTERS};
use std::io::{self,BufRead};
fn main(){for line in io::stdin().lock().lines(){let line=line.unwrap();let p:Vec<_>=line.split_whitespace().collect();let x:i128=p[1].parse().unwrap();let d:i128=p[2].parse().unwrap();let aux:Vec<u32>=if p[0]=="F"{p[3].split(',').map(|n|n.parse().unwrap()).collect()}else{vec![]};let input=if p[0]=="F"{CramCiphertext::wrap_with_fpd_aux((),&[x],None,&aux)}else{CramCiphertext::wrap_default((),&[x],None)};let before=CRAM_DIVISION_COUNTERS.snapshot();let result=match p[0]{"F"=>input.cram_rescale_by_scalar_fpd(d,p[4].parse().unwrap(),|_,_|()),"D"=>input.cram_rescale_by_scalar_div_exact(d,|_,_|()),_=>input.cram_rescale_by_scalar_kelim(d,p[3].parse().unwrap(),|_,_|())};let after=CRAM_DIVISION_COUNTERS.snapshot();print!("{} {} ",after.0-before.0,after.1-before.1);match result{Ok((out,_))=>{print!("OK ");for r in out.witness.c0_signature.signatures[0].residues{print!("{r},");}if let Some(a)=out.witness.c0_aux().as_ref(){print!(" ");for i in 0..aux.len(){print!("{},",a.residue(0,i).unwrap());}}println!()},Err(e)=>println!("E {e:?}")}}}
